cmake_minimum_required(VERSION 3.19)
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/Resources.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(lhdc_decoder LANGUAGES C)
  option(LHDC_BUILD_TESTS "Build decoder public API tests" OFF)
  if(LHDC_BUILD_TESTS)
    enable_testing()
  endif()
endif()
if(NOT TARGET lhdc_common)
  add_subdirectory(../common "${CMAKE_CURRENT_BINARY_DIR}/common")
endif()
if(ANDROID AND ANDROID_PLATFORM_LEVEL LESS 18)
  message(FATAL_ERROR "Supported decoder builds require Android API 18 or newer; use android-21 for both ARM ABIs.")
endif()
if(NOT TARGET lhdc_dsp)
  # Standalone decoder builds consume the same common DSP sources.
  option(LHDC_KISS_FFT_ITERATIVE_STATIC "Use iterative float FFT with static tables" OFF)
  add_library(lhdc_dsp STATIC ../common/src/dsp/dct4_float32.c ../common/src/dsp/dct4_float32_tables.c ../common/src/dsp/fft_float32.c ../common/src/dsp/fft_fixed32.c)
  target_link_libraries(lhdc_dsp PUBLIC audio_math::math)
  lhdc_resource_settings(lhdc_dsp)
  target_compile_definitions(lhdc_dsp PRIVATE
    LHDC_KISS_FFT_ITERATIVE_STATIC=$<BOOL:${LHDC_KISS_FFT_ITERATIVE_STATIC}>)
  add_library(lhdc::dsp ALIAS lhdc_dsp)
  target_include_directories(lhdc_dsp PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/../common/include")
  target_compile_features(lhdc_dsp PRIVATE c_std_11)
  set_target_properties(lhdc_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(MSVC)
    target_compile_options(lhdc_dsp PRIVATE /fp:strict /utf-8 $<$<EQUAL:${CMAKE_SIZEOF_VOID_P},4>:/arch:SSE2>)
  else()
    target_compile_options(lhdc_dsp PRIVATE -ffp-contract=off -fno-fast-math
      -Wunused-variable -Wunused-function -Wunused-const-variable -Wparentheses -Wuninitialized)
    target_link_libraries(lhdc_dsp PUBLIC m)
  endif()
endif()
file(GLOB_RECURSE decoder_v4_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/v4/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/api/v4/*.c")
add_library(lhdc_decoder_v4_objects OBJECT ${decoder_v4_sources})
target_link_libraries(lhdc_decoder_v4_objects PRIVATE audio_math::math)
lhdc_resource_settings(lhdc_decoder_v4_objects)
target_include_directories(lhdc_decoder_v4_objects PRIVATE include src/v4/include src/api/v4/include)
target_compile_features(lhdc_decoder_v4_objects PRIVATE c_std_11)
target_compile_definitions(lhdc_decoder_v4_objects PRIVATE LHDC_ENABLE_LOGGING=$<BOOL:${LHDC_ENABLE_LOGGING}>)
if(MSVC)
  target_compile_options(lhdc_decoder_v4_objects PRIVATE /fp:strict /utf-8
    $<$<EQUAL:${CMAKE_SIZEOF_VOID_P},4>:/arch:SSE2>)
  target_compile_definitions(lhdc_decoder_v4_objects PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(lhdc_decoder_v4_objects PRIVATE -ffp-contract=off
    -Wunused-variable -Wunused-function -Wunused-const-variable -Wparentheses -Wuninitialized)
endif()
add_library(lhdc_decoder_v4 STATIC $<TARGET_OBJECTS:lhdc_decoder_v4_objects>)
add_library(lhdc::decoder_v4 ALIAS lhdc_decoder_v4)
target_link_libraries(lhdc_decoder_v4 PUBLIC lhdc::dsp lhdc::common)
target_include_directories(lhdc_decoder_v4 PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
file(GLOB_RECURSE decoder_v5_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/v5/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/src/api/v5/*.c")
add_library(lhdc_decoder_v5_objects OBJECT ${decoder_v5_sources})
target_link_libraries(lhdc_decoder_v5_objects PRIVATE audio_math::math)
lhdc_resource_settings(lhdc_decoder_v5_objects)
target_include_directories(lhdc_decoder_v5_objects PRIVATE include src/v5/include src/api/v5/include)
target_compile_features(lhdc_decoder_v5_objects PRIVATE c_std_11)
target_compile_definitions(lhdc_decoder_v5_objects PRIVATE LHDC_ENABLE_LOGGING=$<BOOL:${LHDC_ENABLE_LOGGING}>)
if(MSVC)
  target_compile_options(lhdc_decoder_v5_objects PRIVATE /fp:strict /utf-8
    $<$<EQUAL:${CMAKE_SIZEOF_VOID_P},4>:/arch:SSE2>)
  target_compile_definitions(lhdc_decoder_v5_objects PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(lhdc_decoder_v5_objects PRIVATE -ffp-contract=off
    -Wunused-variable -Wunused-function -Wunused-const-variable -Wparentheses -Wuninitialized)
endif()
add_library(lhdc_decoder_v5 STATIC $<TARGET_OBJECTS:lhdc_decoder_v5_objects>)
add_library(lhdc::decoder_v5 ALIAS lhdc_decoder_v5)
target_link_libraries(lhdc_decoder_v5 PUBLIC lhdc::dsp lhdc::common)
target_include_directories(lhdc_decoder_v5 PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
add_library(lhdc_decoder STATIC $<TARGET_OBJECTS:lhdc_decoder_v4_objects> $<TARGET_OBJECTS:lhdc_decoder_v5_objects>)
add_library(lhdc::decoder ALIAS lhdc_decoder)
foreach(target lhdc_decoder_v4 lhdc_decoder_v5 lhdc_decoder)
  target_compile_definitions(${target} INTERFACE LHDC_ENABLE_LOGGING=$<BOOL:${LHDC_ENABLE_LOGGING}>)
endforeach()
target_link_libraries(lhdc_decoder PUBLIC lhdc::dsp lhdc::common)
target_include_directories(lhdc_decoder PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>")
if(UNIX)
  target_link_libraries(lhdc_decoder_v4 PUBLIC m)
  target_link_libraries(lhdc_decoder_v5 PUBLIC m)
  target_link_libraries(lhdc_decoder PUBLIC m)
endif()
if(LHDC_BUILD_TESTS)
  add_subdirectory(tests)
endif()
