if(NOT TARGET lhdc_common)
  add_library(lhdc_common STATIC src/buffer/cirbuf.c src/dsp/adsq_tables.c src/memory/workspace.c ../platform/common/memory.c)
  target_link_libraries(lhdc_common PUBLIC audio_common::core audio_common::math)
  lhdc_resource_settings(lhdc_common)
  add_library(lhdc::common ALIAS lhdc_common)
  target_compile_definitions(lhdc_common PRIVATE LHDC_EXTERNAL_ALLOCATOR_ONLY=$<BOOL:${LHDC_EXTERNAL_ALLOCATOR_ONLY}>)
  target_include_directories(lhdc_common PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_compile_features(lhdc_common PRIVATE c_std_11)
  set_target_properties(lhdc_common PROPERTIES POSITION_INDEPENDENT_CODE ON)
  if(MSVC)
    target_compile_options(lhdc_common PRIVATE /O2 /fp:strict /utf-8)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      target_compile_options(lhdc_common PRIVATE /arch:SSE2)
    endif()
  else()
    target_compile_options(lhdc_common PRIVATE -O2 -ffp-contract=off -fno-fast-math)
    target_link_libraries(lhdc_common PUBLIC m)
    if(ANDROID)
      include("${CMAKE_CURRENT_LIST_DIR}/../platform/android/Android.cmake")
      lhdc_android_arm32_settings(lhdc_common)
    endif()
  endif()
endif()
