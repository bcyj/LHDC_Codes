cmake_minimum_required(VERSION 3.19)
project(lhdc LANGUAGES C)
include(cmake/Resources.cmake)

option(LHDC_BUILD_ENCODER "Build encoder component" ON)
option(LHDC_BUILD_DECODER "Build LHDC V4/V5 and LLAC audio decoders" ON)
option(LHDC_BUILD_AR "Build PCM effects component" ON)
option(LHDC_ENCODER_WITH_AR "Enable AR processing in encoder instances" ${LHDC_BUILD_AR})
option(LHDC_BUILD_BUNDLE "Build encoder and PCM effects shared library" ON)
option(LHDC_BUILD_TESTS "Build component linkage and PCM smoke test" OFF)
option(LHDC_V5_ENCODER_FIXED_FFT "Use integer FFT inside the V5 encoder MDCT" OFF)
option(LHDC_ENCODER_HIGH_PERFORMANCE_SSE
  "Enable faster non-bitexact SSE approximations in the encoder" OFF)
option(LHDC_AR_HIGH_PERFORMANCE_SSE
  "Enable faster non-bitexact SSE processing in the AR PCM effects component" OFF)
set(LHDC_V4_SHIFT_COUNT_BITS "AUTO" CACHE STRING "V4 shift count model: AUTO, 5 (modulo32), or 8 (wide shifts zero/sign)")
set_property(CACHE LHDC_V4_SHIFT_COUNT_BITS PROPERTY STRINGS AUTO 5 8)
if(NOT LHDC_V4_SHIFT_COUNT_BITS MATCHES "^(AUTO|5|8)$")
  message(FATAL_ERROR "LHDC_V4_SHIFT_COUNT_BITS must be AUTO, 5 or 8")
endif()
if(LHDC_ENCODER_WITH_AR AND NOT LHDC_BUILD_AR)
  message(FATAL_ERROR "LHDC_ENCODER_WITH_AR requires LHDC_BUILD_AR")
endif()

if(LHDC_BUILD_TESTS)
  enable_testing()
endif()
set(LHDC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/lhdc")
set(AUDIO_COMMON_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../audio_codec_common" CACHE PATH
  "Path to the independent audio_codec_common source tree")
include(cmake/Components.cmake)
if(LHDC_BUILD_TESTS)
  set(AUDIO_COMMON_BUILD_TESTS ON)
  set(AUDIO_MATH_BUILD_TESTS ON)
endif()
set(AUDIO_COMMON_NO_CRT ${LHDC_DRIVER_RESOURCES})
set(AUDIO_COMMON_DRIVER_RESOURCES ${LHDC_DRIVER_RESOURCES})
if(LHDC_AR_HIGH_PERFORMANCE_SSE)
  set(AUDIO_MATH_FFT_SPLIT_RADIX_SSE ON)
endif()
if(NOT TARGET audio_common::core OR NOT TARGET audio_common::math)
  if(NOT EXISTS "${AUDIO_COMMON_ROOT}/CMakeLists.txt")
    message(FATAL_ERROR
      "audio_codec_common was not found at ${AUDIO_COMMON_ROOT}; set AUDIO_COMMON_ROOT explicitly")
  endif()
  add_subdirectory("${AUDIO_COMMON_ROOT}" audio_codec_common)
endif()
add_subdirectory(lhdc/common common)
add_library(lhdc_platform STATIC)
set_target_properties(lhdc_platform PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  target_sources(lhdc_platform PRIVATE lhdc/platform/windows/windows_codec_math.c
    lhdc/platform/windows/windows_codec_math_tables.c lhdc/platform/windows/windows_sine.c lhdc/platform/common/silent_log.c)
  target_include_directories(lhdc_platform PRIVATE lhdc/platform/windows/include)
else()
  # Android uses native logd; other user-mode platforms use the silent backend.
  target_sources(lhdc_platform PRIVATE lhdc/common/src/platform_anchor.c)
  target_link_libraries(lhdc_platform PUBLIC m)
  if(ANDROID)
    lhdc_android_platform(lhdc_platform)
  else()
    target_sources(lhdc_platform PRIVATE lhdc/platform/common/silent_log.c)
  endif()
endif()
lhdc_compile_settings(lhdc_platform)
lhdc_resource_settings(lhdc_platform)
target_link_libraries(lhdc_platform PUBLIC lhdc::common)
target_compile_definitions(lhdc_platform PUBLIC LHDC_ENABLE_LOGGING=$<BOOL:${LHDC_ENABLE_LOGGING}>)

add_library(lhdc_dsp_objects OBJECT lhdc/common/src/dsp/fft_float32.c lhdc/common/src/dsp/dct4_float32.c lhdc/common/src/dsp/dct4_float32_tables.c lhdc/common/src/dsp/fft_fixed32.c)
lhdc_compile_settings(lhdc_dsp_objects)
lhdc_resource_settings(lhdc_dsp_objects)
target_compile_definitions(lhdc_dsp_objects PRIVATE
  LHDC_KISS_FFT_ITERATIVE_STATIC=$<BOOL:${LHDC_KISS_FFT_ITERATIVE_STATIC}>)
set_target_properties(lhdc_dsp_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
  lhdc_android_compile_settings(lhdc_dsp_objects)
endif()
add_library(lhdc_dsp STATIC $<TARGET_OBJECTS:lhdc_dsp_objects>)
add_library(lhdc::dsp ALIAS lhdc_dsp)
target_include_directories(lhdc_dsp PUBLIC "$<BUILD_INTERFACE:${LHDC_ROOT}/common/include>")
target_link_libraries(lhdc_dsp PUBLIC lhdc_platform)
if(LHDC_BUILD_DECODER)
  add_subdirectory(lhdc/decoder decoder)
endif()
if(LHDC_BUILD_AR)
  set(AUDIO_AR_BUILD_TESTS ${LHDC_BUILD_TESTS})
  add_subdirectory(ar public_ar)
  add_subdirectory(lhdc/ar ar)
endif()
if(LHDC_BUILD_ENCODER)
  add_subdirectory(lhdc/encoder encoder)
endif()
if(LHDC_BUILD_BUNDLE AND LHDC_BUILD_ENCODER AND LHDC_BUILD_AR AND LHDC_ENCODER_WITH_AR)
  add_library(lhdc_all SHARED $<TARGET_OBJECTS:lhdc_dsp_objects> $<TARGET_OBJECTS:lhdc_encoder_v4_objects>
    $<TARGET_OBJECTS:lhdc_encoder_v5_objects> $<TARGET_OBJECTS:lhdc_ar_v4_objects>
    $<TARGET_OBJECTS:lhdc_ar_v5_objects> $<TARGET_OBJECTS:lhdc_ar_common_objects>)
  target_link_libraries(lhdc_all PRIVATE lhdc_platform lhdc_common)
  if(MSVC)
    target_sources(lhdc_all PRIVATE lhdc/platform/windows/lhdc_all.def)
  elseif(WIN32)
    target_link_options(lhdc_all PRIVATE -Wl,--export-all-symbols)
  else()
    target_link_options(lhdc_all PRIVATE -Wl,-Bsymbolic -Wl,--no-undefined)
  endif()
endif()
if(LHDC_BUILD_TESTS)
  if(WIN32)
    add_executable(lhdc_windows_math_failure lhdc/common/tests/windows_math_failure_test.c)
    lhdc_compile_settings(lhdc_windows_math_failure)
    add_test(NAME lhdc_windows_math_failure COMMAND lhdc_windows_math_failure)
  endif()
  add_executable(lhdc_logging_api lhdc/common/tests/logging_test.c)
  lhdc_compile_settings(lhdc_logging_api)
  target_link_libraries(lhdc_logging_api PRIVATE lhdc_platform)
  add_test(NAME lhdc_logging_api COMMAND lhdc_logging_api)
  # Build a private recursive reference with isolated symbols for strict backend comparison.
  add_library(lhdc_fft_float32_reference_objects OBJECT
    "${AUDIO_COMMON_ROOT}/math/src/fft_float32.c")
  lhdc_compile_settings(lhdc_fft_float32_reference_objects)
  target_compile_definitions(lhdc_fft_float32_reference_objects PRIVATE
    AUDIO_MATH_FFT_ITERATIVE_STATIC=0
    AUDIO_MATH_FFT_512_BITEXACT_SSE2=0
    AUDIO_MATH_FFT_BITEXACT_SSE2=0
    LHDC_KISS_FFT_ITERATIVE_STATIC=0
    audio_math_fft_work=lhdc_reference_fft_work
    audio_math_fft_factor=lhdc_reference_fft_factor
    audio_math_fft_twiddles=lhdc_reference_fft_twiddles
    audio_math_fftr_copy_twiddles=lhdc_reference_fftr_copy_twiddles
    audio_math_fft_next_fast_size=lhdc_reference_fft_next_fast_size
    lhdc_kiss_fft_split=lhdc_reference_fft_split
    audio_math_fft_512=lhdc_reference_fft_512)
  if(ANDROID)
    lhdc_android_arm32_settings(lhdc_fft_float32_reference_objects)
  endif()
  add_executable(lhdc_fft_float32_backend_api lhdc/common/tests/fft_float32_backend_test.c
    $<TARGET_OBJECTS:lhdc_fft_float32_reference_objects>)
  lhdc_compile_settings(lhdc_fft_float32_backend_api)
  target_link_libraries(lhdc_fft_float32_backend_api PRIVATE lhdc::dsp)
  add_test(NAME lhdc_fft_float32_backend_api COMMAND lhdc_fft_float32_backend_api)
  add_executable(lhdc_dct4_public_api lhdc/common/tests/dct4_float32_test.c)
  lhdc_compile_settings(lhdc_dct4_public_api)
  target_link_libraries(lhdc_dct4_public_api PRIVATE lhdc::dsp)
  add_test(NAME lhdc_dct4_public_api COMMAND lhdc_dct4_public_api)
  if(NOT TARGET lhdc_fft_fixed32_public_api)
    add_executable(lhdc_fft_fixed32_public_api lhdc/common/tests/fft_fixed32_test.c)
    lhdc_compile_settings(lhdc_fft_fixed32_public_api)
    target_link_libraries(lhdc_fft_fixed32_public_api PRIVATE lhdc::dsp)
    add_test(NAME lhdc_fft_fixed32_public_api COMMAND lhdc_fft_fixed32_public_api)
  endif()
  add_subdirectory(tests/integration)
endif()
