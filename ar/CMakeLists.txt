cmake_minimum_required(VERSION 3.19)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(audio_ar_interface LANGUAGES C)
endif()
set(AR_DEFAULT_BACKEND "LHDC" CACHE STRING "Default PCM effect backend: LHDC or NONE")
set_property(CACHE AR_DEFAULT_BACKEND PROPERTY STRINGS LHDC NONE)
option(AUDIO_AR_BUILD_TESTS "Build the generic backend contract test" OFF)
add_library(audio_ar INTERFACE)
add_library(audio_ar::ar ALIAS audio_ar)
target_include_directories(audio_ar INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
if(AR_DEFAULT_BACKEND STREQUAL "LHDC")
  if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "Use the repository root for the default LHDC backend, or set AR_DEFAULT_BACKEND=NONE for the standalone interface")
  endif()
  target_link_libraries(audio_ar INTERFACE lhdc::ar)
  target_compile_definitions(audio_ar INTERFACE AR_HAS_DEFAULT_BACKEND=1
    AR_DEFAULT_BACKEND_PROVIDER=lhdc_ar_default_backend)
elseif(AR_DEFAULT_BACKEND STREQUAL "NONE")
  target_compile_definitions(audio_ar INTERFACE AR_HAS_DEFAULT_BACKEND=0)
else()
  message(FATAL_ERROR "AR_DEFAULT_BACKEND must be LHDC or NONE")
endif()
if(AUDIO_AR_BUILD_TESTS)
  enable_testing()
  add_executable(audio_ar_backend_contract tests/backend_contract.c)
  target_compile_features(audio_ar_backend_contract PRIVATE c_std_11)
  target_include_directories(audio_ar_backend_contract PRIVATE include)
  target_compile_definitions(audio_ar_backend_contract PRIVATE AR_HAS_DEFAULT_BACKEND=0)
  add_test(NAME audio_ar_backend_contract COMMAND audio_ar_backend_contract)
endif()
